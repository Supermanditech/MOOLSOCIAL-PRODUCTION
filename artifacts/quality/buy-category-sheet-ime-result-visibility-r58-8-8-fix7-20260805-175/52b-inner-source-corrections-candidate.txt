$adbReplacement = @'
function Invoke-AuditAdb {
  param(
    [Parameter(Position = 0, ValueFromRemainingArguments = $true)]
    [string[]]$Arguments
  )
  $previousPreference = $ErrorActionPreference
  $ErrorActionPreference = 'Continue'
  try {
    $output = & adb -s $Serial @Arguments 2>&1
    $nativeExit = $LASTEXITCODE
  } finally {
    $ErrorActionPreference = $previousPreference
  }
  if ($nativeExit -ne 0) {
    throw "adb failed: $($Arguments -join ' ')`n$($output -join "`n")"
  }
  return $output
}
'@
$adbRegex = [regex]::new('(?ms)^function Invoke-AuditAdb \{.*?^\}')
if ($adbRegex.Matches($source).Count -ne 1) {
  throw 'Expected one Invoke-AuditAdb function.'
}
$source = $adbRegex.Replace(
  $source,
  [System.Text.RegularExpressions.MatchEvaluator]{
    param($match)
    return $adbReplacement
  },
  1
)
